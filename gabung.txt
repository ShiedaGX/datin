<?php
namespace App\Http\Controllers;

use App\Http\Requests\StoreWilayahRequest;
use App\Models\DataPemda;
use App\Models\DokumenPengajuan;
use App\Models\KabKota;
use App\Models\Moda;
use App\Models\PermintaanRevisi;
use App\Models\Provinsi;
use App\Models\RuteSubsidi;
use App\Models\Wilayah;
use App\Support\FormOptions;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\ValidationException;

class InputDataController extends Controller
{
    public function create(Request $request)
    {
        $user = auth()->user();

        // id_wilayah pada provinsi ikut diambil supaya Wilayah Regional bisa terisi otomatis
        $semua = KabKota::with('provinsi:id_provinsi,id_wilayah,nama_provinsi', 'dataPemda.ruteSubsidi', 'dataPemda.permintaanRevisi')
            ->orderBy('nama_kab_kota')->get();

        // Kab/kota milik akun (Pemda): wilayah, provinsi, dan kab/kota terisi otomatis & terkunci
        $kabSaya = $user->id_kab_kota ? $semua->firstWhere('id_kab_kota', (int) $user->id_kab_kota) : null;
        $terkunci = $kabSaya !== null;
        $akunBelumTerhubung = ! $terkunci && ! $user->isSuperAdmin();

        [$isian, $sumber] = $this->isianDariPengajuanSebelumnya($request, $semua, $user);
        $namaModa = Moda::pluck('nama_moda', 'id_moda');

        return view('input-data.create', [
            'isian'       => $isian,           // data pengajuan sebelumnya (revisi INTRAM), kosong untuk pengajuan baru
            'sumber'      => $sumber,
            'terkunci'    => $terkunci,
            'kabSaya'     => $kabSaya,
            'alasanSaya'  => $kabSaya?->alasanTidakTersedia($user),
            'akunBelumTerhubung' => $akunBelumTerhubung,
            // saran sektor pada kolom Potensi Unggulan Daerah (nama + kata kunci untuk pencarian)
            'sektor'      => collect(\App\Support\Sektor::peta())->map(fn ($kunci, $nama) => ['nama' => $nama, 'kunci' => $kunci])->values(),
            'provinsi'    => Provinsi::orderBy('nama_provinsi')->get(),
            'wilayah'     => Wilayah::orderBy('nama_wilayah')->get(),
            'moda'        => Moda::orderBy('id_moda')->get(),
            'klasifikasi' => FormOptions::KLASIFIKASI,
            'keterangan'  => FormOptions::KETERANGAN,
            'dokumen'     => FormOptions::DOKUMEN,
            // pilihan dropdown Kabupaten/Kota
            'kabKota'     => $semua->map(fn ($k) => [
                'id'          => $k->id_kab_kota,
                'id_provinsi' => $k->id_provinsi,
                'nama'        => $k->nama_kab_kota,
                'alasan'      => $k->alasanTidakTersedia($user),
                'revisi'      => $k->modaRevisiTerbuka($user),     // id moda yang diminta INTRAM
                'sudah'       => $k->modaDiajukan($user),          // id moda yang sudah diajukan akun ini
                'sudahNama'   => collect($k->modaDiajukan($user))->map(fn ($id) => $namaModa[$id] ?? '')->filter()->values()->all(),
                'revisiNama'  => Moda::whereIn('id_moda', $k->modaRevisiTerbuka($user))->orderBy('id_moda')->pluck('nama_moda')->all(),
            ])->values(),
            // saran otomatis untuk kolom Kebutuhan Konektivitas
            'saran'       => $semua->map(fn ($k) => [
                'id'       => $k->id_kab_kota,
                'nama'     => $k->nama_kab_kota,
                'provinsi' => $k->provinsi->nama_provinsi,
            ])->values(),
        ]);
    }

    /**
     * Pengajuan ulang atas permintaan revisi INTRAM (?kab=ID): form diisi dengan data pengajuan terakhir
     * daerah tersebut agar Pemda cukup memeriksa dan mengubah seperlunya. Dokumen tetap harus diunggah.
     *
     * @return array{0: array, 1: ?RuteSubsidi}
     */
    private function isianDariPengajuanSebelumnya(Request $request, $semuaKabKota, $user): array
    {
        $idKab = (int) $request->query('kab');

        // Akun yang terhubung ke satu kab/kota hanya boleh memakai kab/kota miliknya
        if ($idKab && $user->id_kab_kota && (int) $user->id_kab_kota !== $idKab) {
            return [[], null];
        }

        $kab = $idKab ? $semuaKabKota->firstWhere('id_kab_kota', $idKab) : null;
        if (! $kab) {
            return [[], null];
        }

        $modaDiminta = $kab->modaRevisiTerbuka($user);
        $sumber = $modaDiminta ? RuteSubsidi::where('id_kab_kota_asal', $idKab)->orderByDesc('id_rute_subsidi')->first() : null;
        if (! $sumber) {
            return [[], null];
        }

        $kunciKeterangan = collect(FormOptions::KETERANGAN)->mapWithKeys(fn ($v, $k) => [mb_strtolower($v['label']) => $k]);

        return [[
            'id_wilayah'             => Provinsi::where('id_provinsi', $kab->id_provinsi)->value('id_wilayah'),
            'id_provinsi'            => $kab->id_provinsi,
            'id_kab_kota'            => $kab->id_kab_kota,
            'klasifikasi_pemda'      => $sumber->klasifikasi_pemda,
            'keterangan'             => collect($sumber->keterangan_daftar)->map(fn ($l) => $kunciKeterangan[mb_strtolower($l)] ?? null)->filter()->values()->all(),
            'potensi'                => array_slice($sumber->potensi_daftar, 0, 5),
            'kebutuhan_konektivitas' => $sumber->kebutuhan_konektivitas,
            'id_moda'                => count($modaDiminta) === 1 ? $modaDiminta[0] : null,
        ], $sumber];
    }

    public function store(StoreWilayahRequest $request)
    {
        $data = $request->validated();
        $user = $request->user();

        // Pengaman: kab/kota harus milik akun (readonly di browser bisa diakali lewat DevTools)
        if ($user->id_kab_kota) {
            if ((int) $data['id_kab_kota'] !== (int) $user->id_kab_kota) {
                throw ValidationException::withMessages(['id_kab_kota' => 'Kabupaten/Kota harus sesuai dengan akun Anda.']);
            }
        } elseif (! $user->isSuperAdmin()) {
            throw ValidationException::withMessages(['id_kab_kota' => 'Akun Anda belum terhubung ke kabupaten/kota. Hubungi Super Admin.']);
        }

        $kab = KabKota::findOrFail($data['id_kab_kota']);

        // Format sama dengan data Excel: "1. A; 2. B; 3. C"
        $potensi = collect($data['potensi'])
            ->map(fn ($v) => trim((string) $v))
            ->filter()
            ->values()
            ->map(fn ($v, $i) => ($i + 1) . '. ' . $v)
            ->implode('; ');

        // Format sama dengan data Excel: "3TP, KSPEAN, ..."
        $keterangan = collect($data['keterangan'] ?? [])
            ->map(fn ($key) => FormOptions::KETERANGAN[$key]['nilai'])
            ->implode(', ');

        // "Merauke, Jayapura, " -> "Merauke, Jayapura"
        $konektivitas = trim((string) ($data['kebutuhan_konektivitas'] ?? ''), " \t\r\n,");

        $tersimpan = []; // path file, untuk dibersihkan jika transaksi gagal

        try {
            DB::transaction(function () use ($request, $data, $user, $kab, $potensi, $keterangan, $konektivitas, &$tersimpan) {
                // Daerah dipilih dari data yang sudah ada: isi/ambil alih data_pemda miliknya.
                $pemda = DataPemda::where('id_kab_kota', $kab->id_kab_kota)->lockForUpdate()->first()
                    ?? new DataPemda(['id_kab_kota' => $kab->id_kab_kota]);

                // Pemda boleh mengajukan berkali-kali (pengajuan lama tidak diubah); data milik akun lain tetap ditolak.
                $permintaan = null;
                if ($pemda->exists) {
                    if (! $pemda->bisaDiajukanOleh($user)) {
                        throw ValidationException::withMessages(['id_kab_kota' => 'Kabupaten/Kota ini sudah diajukan atau dikelola akun lain.']);
                    }
                    // Bila INTRAM sedang meminta revisi untuk moda ini, pengajuan ini otomatis memenuhinya
                    $permintaan = PermintaanRevisi::where('id_pemda', $pemda->id_pemda)
                        ->where('id_moda', $data['id_moda'])->where('status', PermintaanRevisi::MENUNGGU)
                        ->lockForUpdate()->first();
                }

                $isian = [
                    'potensi_pemda'          => $potensi,
                    'klasifikasi_pemda'      => $data['klasifikasi_pemda'],
                    'keterangan_pemda'       => $keterangan !== '' ? $keterangan : null,
                    'kebutuhan_konektivitas' => $konektivitas !== '' ? $konektivitas : null,
                ];

                // data_pemda menyimpan isian terbaru daerah; setiap pengajuan juga menyimpan salinannya sendiri
                $pemda->fill(['id_user' => $user->id_user] + $isian)->save();

                // Satu pengajuan = satu rute_subsidi (satu moda). Kota tujuan tertulis pada kebutuhan_konektivitas.
                $rute = RuteSubsidi::create([
                    'id_pemda'           => $pemda->id_pemda,
                    'id_moda'            => $data['id_moda'],
                    'id_kab_kota_asal'   => $kab->id_kab_kota,
                    'id_kab_kota_tujuan' => null,
                    'status_pengajuan'   => FormOptions::STATUS_AWAL,
                    'tanggal_pengajuan'  => now()->toDateString(),
                ] + $isian);

                if ($permintaan) {
                    $permintaan->update(['status' => PermintaanRevisi::DIPENUHI, 'id_rute_baru' => $rute->id_rute_subsidi]);
                }

                foreach (FormOptions::DOKUMEN as $field => $jenis) {
                    $file = $request->file($field);
                    $path = $file->store("dokumen_pengajuan/{$rute->id_rute_subsidi}", 'local');
                    $tersimpan[] = $path;

                    DokumenPengajuan::create([
                        'id_rute_subsidi' => $rute->id_rute_subsidi,
                        'id_user'         => $user->id_user,
                        'jenis_dokumen'   => $jenis,
                        'nama_file'       => $file->getClientOriginalName(),
                        'nomor_surat'     => trim($data['nomor'][$field]),
                        'path_file'       => $path,
                        'versi_dokumen'   => 1,
                        'tanggal_upload'  => now(),
                    ]);
                }
            });
        } catch (ValidationException $e) {
            Storage::disk('local')->delete($tersimpan);

            throw $e;
        } catch (\Throwable $e) {
            Storage::disk('local')->delete($tersimpan);
            report($e);

            return back()->withInput()->with('error', 'Data gagal disimpan. Silakan coba lagi atau hubungi administrator.');
        }

        return redirect()->route('data-wilayah.index')
            ->with('success', 'Pengajuan untuk "' . $kab->nama_kab_kota . '" berhasil ditambahkan.');
    }
}

===== app\Http\Requests\StoreWilayahRequest.php =====
<?php
namespace App\Http\Requests;

use App\Models\KabKota;
use App\Models\Provinsi;
use App\Support\FormOptions;
use Illuminate\Foundation\Http\FormRequest;
use Illuminate\Validation\Rule;

class StoreWilayahRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true; // akses dibatasi middleware 'auth' pada route
    }

    public function rules(): array
    {
        $rules = [
            'id_provinsi'            => ['required', 'integer', 'exists:provinsi,id_provinsi'],
            'id_wilayah'             => ['required', 'integer', 'exists:wilayah,id_wilayah'],
            'id_kab_kota'            => ['required', 'integer', 'exists:kab_kota,id_kab_kota'],
            'klasifikasi_pemda'      => ['required', Rule::in(FormOptions::KLASIFIKASI)],
            'keterangan'             => ['nullable', 'array'],
            'keterangan.*'           => ['string', Rule::in(array_keys(FormOptions::KETERANGAN))],
            'potensi'                => ['required', 'array', 'max:5'],
            'potensi.*'              => ['nullable', 'string', 'max:200', function ($attribute, $value, $fail) {
                // satu potensi per kolom; pengecualian: nama sektor dari daftar saran yang memang memuat koma
                if ($value !== null && preg_match('/[,;\r\n]/', (string) $value) && ! in_array(trim((string) $value), \App\Support\Sektor::daftar(), true)) {
                    $fail('Isi satu potensi saja pada setiap kolom (tanpa koma, titik koma, atau baris baru).');
                }
            }],
            'id_moda'                => ['required', 'integer', 'exists:moda,id_moda'],
            // PERUBAHAN: menggantikan rule 'kebutuhan_konektivitas'
            'tujuan'                 => ['required', 'array', 'min:1', 'max:20'],
            'tujuan.*'               => ['integer', 'distinct', 'exists:kab_kota,id_kab_kota', 'different:id_kab_kota'],
        ];

        foreach (array_keys(FormOptions::DOKUMEN) as $field) {
            $rules[$field] = ['required', 'file', 'mimes:pdf,doc,docx', 'max:5120']; // 5 MB
            $rules['nomor.' . $field] = ['required', 'string', 'max:100'];          // nomor surat diisi manual
        }

        return $rules;
    }

    public function withValidator($validator): void
    {
        $validator->after(function ($v) {
            // Provinsi harus berada di wilayah regional yang dipilih
            if (! $v->errors()->hasAny(['id_provinsi', 'id_wilayah'])) {
                $provinsi = Provinsi::find($this->input('id_provinsi'));
                if ($provinsi && (int) $provinsi->id_wilayah !== (int) $this->input('id_wilayah')) {
                    $v->errors()->add('id_wilayah', 'Wilayah regional tidak sesuai dengan provinsi yang dipilih.');
                }
            }

            // Kab/Kota harus berada di provinsi yang dipilih dan masih tersedia
            if (! $v->errors()->hasAny(['id_provinsi', 'id_kab_kota'])) {
                $kab = KabKota::with('dataPemda.ruteSubsidi', 'dataPemda.permintaanRevisi')->find($this->input('id_kab_kota'));
                if ($kab && (int) $kab->id_provinsi !== (int) $this->input('id_provinsi')) {
                    $v->errors()->add('id_kab_kota', 'Kabupaten/Kota tidak berada pada provinsi yang dipilih.');
                } elseif ($kab && ($alasan = $kab->alasanTidakTersedia($this->user()))) {
                    $v->errors()->add('id_kab_kota', 'Kabupaten/Kota ini ' . $alasan . '.');
                }
            }

            // Minimal satu potensi unggulan
            $terisi = collect($this->input('potensi', []))->filter(fn ($x) => trim((string) $x) !== '');
            if ($terisi->isEmpty() && ! $v->errors()->has('potensi')) {
                $v->errors()->add('potensi', 'Isi minimal satu potensi unggulan daerah.');
            }
        });
    }

    public function attributes(): array
    {
        $nomor = [];
        foreach (FormOptions::DOKUMEN as $field => $label) {
            $nomor['nomor.' . $field] = 'Nomor ' . mb_strtolower($label);
        }

        return $nomor + [
            'id_provinsi'            => 'Provinsi',
            'id_wilayah'             => 'Wilayah regional',
            'id_kab_kota'            => 'Kabupaten/Kota',
            'klasifikasi_pemda'      => 'Klasifikasi wilayah',
            'id_moda'                => 'Moda transportasi',
            // PERUBAHAN: menggantikan atribut 'kebutuhan_konektivitas'
            'tujuan'                 => 'Rute tujuan',
            'tujuan.*'               => 'Rute tujuan',
            'potensi.*'              => 'Potensi unggulan',
            'surat_permohonan'       => 'Surat permohonan',
            'surat_pernyataan'       => 'Surat pernyataan',
            'surat_rencana_teknis'   => 'Surat rencana teknis',
        ];
    }

    public function messages(): array
    {
        return [
            'required'            => ':attribute wajib diisi.',
            'id_kab_kota.required' => 'Kabupaten/Kota wajib dipilih.',
            'exists'              => ':attribute tidak valid.',
            'in'                  => ':attribute tidak valid.',
            'max.string'          => ':attribute maksimal :max karakter.',
            'max.file'            => ':attribute maksimal 5 MB.',
            'mimes'               => ':attribute harus berformat PDF, DOC, atau DOCX.',
            'uploaded'            => ':attribute gagal diunggah. Periksa batas ukuran upload pada server (upload_max_filesize).',
            'potensi.*.not_regex' => 'Isi satu potensi saja pada setiap kolom (tanpa koma, titik koma, atau baris baru).',
            // PERUBAHAN: pesan untuk validasi tujuan
            'tujuan.required'     => 'Pilih minimal satu rute tujuan.',
            'tujuan.min'          => 'Pilih minimal satu rute tujuan.',
            'tujuan.max'          => 'Rute tujuan maksimal :max kabupaten/kota.',
            'tujuan.*.distinct'   => 'Rute tujuan tidak boleh dipilih lebih dari sekali.',
            'tujuan.*.different'  => 'Rute tujuan tidak boleh sama dengan Kabupaten/Kota asal.',
        ];
    }
}

<?php
namespace App\Models;

use App\Models\Concerns\DataDaerah;
use App\Support\StatusPengajuan;
use App\Support\SuratOptions;
use App\Support\Tanggal;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\Relations\HasManyThrough;
use Illuminate\Database\Eloquent\Relations\HasOne;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;

class RuteSubsidi extends Model
{
    use DataDaerah;

    protected $table = 'rute_subsidi';
    protected $primaryKey = 'id_rute_subsidi';
    public $timestamps = true;
    protected $casts = ['tanggal_pengajuan' => 'date'];
    protected $fillable = ['id_pemda', 'id_moda', 'id_kab_kota_asal', 'id_kab_kota_tujuan', 'status_pengajuan', 'tanggal_pengajuan', 'klasifikasi_pemda', 'keterangan_pemda', 'potensi_pemda', 'kebutuhan_konektivitas'];

    public function pemda(): BelongsTo { return $this->belongsTo(DataPemda::class, 'id_pemda', 'id_pemda'); }
    public function moda(): BelongsTo { return $this->belongsTo(Moda::class, 'id_moda', 'id_moda'); }
    public function kabKotaAsal(): BelongsTo { return $this->belongsTo(KabKota::class, 'id_kab_kota_asal', 'id_kab_kota'); }
    public function kabKotaTujuan(): BelongsTo { return $this->belongsTo(KabKota::class, 'id_kab_kota_tujuan', 'id_kab_kota'); }

    public function tujuan(): BelongsToMany
    {
        return $this->belongsToMany(KabKota::class, 'rute_subsidi_tujuan', 'id_rute_subsidi', 'id_kab_kota_tujuan')
            ->withTimestamps()
            ->orderBy('nama_kab_kota');
    }

    public function verifikasi(): HasMany { return $this->hasMany(VerifikasiPengajuan::class, 'id_rute_subsidi', 'id_rute_subsidi'); }
    public function pengajuanSurat(): HasMany { return $this->hasMany(PengajuanSurat::class, 'id_rute_subsidi', 'id_rute_subsidi'); }
    public function rekomendasi(): HasManyThrough
    {
        return $this->hasManyThrough(RekomendasiTeknis::class, PengajuanSurat::class, 'id_rute_subsidi', 'id_pengajuan_surat', 'id_rute_subsidi', 'id_pengajuan_surat');
    }
    public function dokumen(): HasMany { return $this->hasMany(DokumenPengajuan::class, 'id_rute_subsidi', 'id_rute_subsidi'); }

    // ---- Tampilan halaman Permohonan ----

    /** Contoh: LP-2026-00482 (tahun pengajuan + id laporan). */
    public function getNomorLaporanAttribute(): string
    {
        $waktu = $this->created_at ?? $this->tanggal_pengajuan;

        return 'LP-' . ($waktu ? $waktu->format('Y') : date('Y')) . '-' . str_pad((string) $this->id_rute_subsidi, 5, '0', STR_PAD_LEFT);
    }

    public function getWaktuPengajuanAttribute()
    {
        return $this->created_at ?? $this->tanggal_pengajuan;
    }

    public function getTanggalLengkapAttribute(): string
    {
        return Tanggal::panjang($this->waktu_pengajuan);
    }

    public function getJamPengajuanAttribute(): string
    {
        return $this->created_at ? Tanggal::jam($this->created_at) : '';
    }

    /** menunggu | disetujui | revisi | ditolak | lainnya */
    public function getGrupStatusAttribute(): string
    {
        return StatusPengajuan::grup($this->status_pengajuan);
    }

    public function getSedangMenungguAttribute(): bool
    {
        return $this->grup_status === 'menunggu';
    }

    public function getKelasStatusModaAttribute(): string
    {
        return 's-' . $this->grup_status;
    }

    /** Label status untuk Moda: "Verifikasi 1" dan "Verifikasi Final" ditampilkan sebagai "Disetujui". */
    public function getLabelStatusModaAttribute(): string
    {
        return StatusPengajuan::LABEL_GRUP[$this->grup_status] ?? $this->status_pengajuan;
    }

    // ---- Surat CHR yang terhubung dengan laporan ini ----

    /** Surat CHR terbaru untuk laporan ini (relasi pengajuanSurat sebaiknya sudah di-eager-load). */
    public function getChrTerbaruAttribute(): ?PengajuanSurat
    {
        return $this->pengajuanSurat->sortByDesc('id_pengajuan_surat')->first();
    }

    /** Moda masih perlu/bisa mengunggah CHR: belum ada, masih draf, atau perlu perbaikan. */
    public function getPerluChrAttribute(): bool
    {
        $chr = $this->chr_terbaru;

        return ! $chr || in_array($chr->status, [SuratOptions::DRAF, SuratOptions::PERLU_PERBAIKAN, SuratOptions::DITOLAK], true);
    }

    /** Waktu laporan disetujui (keputusan "Disetujui" terakhir pada riwayat peninjauan). */
    public function getTanggalDisetujuiAttribute()
    {
        $v = $this->verifikasi->where('status_verifikasi', 'Disetujui')->sortByDesc('id_verifikasi_pengajuan')->first();

        return $v ? $v->tanggal_verifikasi : $this->updated_at;
    }

    /** Surat Rekomendasi Teknis terbaru dari INTRAM untuk laporan ini (relasi rekomendasi sebaiknya sudah di-eager-load). */
    public function getRekomendasiTerbaruAttribute(): ?RekomendasiTeknis
    {
        return $this->rekomendasi->sortByDesc('id_rekomendasi_teknis')->first();
    }

    /** [label, kelas CSS] posisi Surat Rekomendasi Teknis laporan ini (relasi pengajuanSurat dan rekomendasi di-eager-load). */
    public function getStatusRekomendasiAttribute(): array
    {
        if ($this->rekomendasi_terbaru) {
            return ['Terunggah', 'sr-disetujui'];
        }
        if ($this->pengajuanSurat->contains('status', SuratOptions::DISETUJUI)) {
            return ['Menunggu Rekomendasi Teknis', 'sr-dalam-review'];
        }
        $chr = $this->chr_terbaru;

        return $chr ? ['CHR ' . $chr->status, 'sr-draf'] : ['Belum ada CHR', 'sr-belum'];
    }
}

<?php

namespace App\Support;

use App\Models\MasterSektor;
use Illuminate\Support\Facades\Cache;

class Sektor
{
    public const CACHE_KEY = 'master_sektor.peta';

    /** @return array<string, string[]> 'Nama Sektor' => [kata kunci...] */
    public static function peta(): array
    {
        return Cache::rememberForever(self::CACHE_KEY, function () {
            return MasterSektor::with('komoditas:id_komoditas,id_sektor,nama_komoditas')
                ->orderBy('urutan')->get()
                ->mapWithKeys(function ($s) {
                    // kata dari nama sektor ("Perikanan & Kelautan" => perikanan, kelautan) + semua komoditas
                    $dariNama = preg_split('/\s*[&,]\s*/', mb_strtolower($s->nama_sektor));
                    $kunci = collect($dariNama)
                        ->merge($s->komoditas->pluck('nama_komoditas')->map(fn ($k) => mb_strtolower($k)))
                        ->filter()->unique()->values()->all();

                    return [$s->nama_sektor => $kunci];
                })->all();
        });
    }

    /** @return string[] daftar nama sektor, sesuai urutan master */
    public static function daftar(): array
    {
        return array_keys(self::peta());
    }

    /**
     * Nama sektor yang terdeteksi dari teks potensi: dihitung bila minimal satu
     * kata kunci ditemukan (tanpa membedakan huruf besar/kecil).
     *
     * @return string[]
     */
    public static function deteksi(string $teks): array
    {
        $teks = mb_strtolower($teks);
        if (trim($teks) === '') {
            return [];
        }

        $hasil = [];
        foreach (self::peta() as $nama => $kunci) {
            foreach ($kunci as $k) {
                if ($k !== '' && str_contains($teks, $k)) {
                    $hasil[] = $nama;
                    break;
                }
            }
        }

        return $hasil;
    }

    public static function lupakan(): void
    {
        Cache::forget(self::CACHE_KEY);
    }
}

<?php
namespace App\Models;

use App\Support\StatusPengajuan;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;

class DataPemda extends Model
{
    protected $table = 'data_pemda';
    protected $primaryKey = 'id_pemda';
    public $timestamps = false;
    protected $fillable = [
        'id_user', 'id_kab_kota', 'potensi_pemda', 'nama_pemda', 'kontak_pemda',
        'klasifikasi_pemda', 'keterangan_pemda', 'kebutuhan_konektivitas',
    ];

    public function user(): BelongsTo { return $this->belongsTo(User::class, 'id_user', 'id_user'); }
    public function kabKota(): BelongsTo { return $this->belongsTo(KabKota::class, 'id_kab_kota', 'id_kab_kota'); }
    public function permintaanRevisi(): HasMany { return $this->hasMany(PermintaanRevisi::class, 'id_pemda', 'id_pemda'); }
    public function ruteSubsidi(): HasMany { return $this->hasMany(RuteSubsidi::class, 'id_pemda', 'id_pemda'); }

    /** Prioritas Strategis | Prioritas | Strategis | Belum diklasifikasi */
    public static function normalisasiKlasifikasi(?string $nilai): string
    {
        $v = mb_strtolower((string) $nilai);
        $p = str_contains($v, 'prioritas');
        $s = str_contains($v, 'strategis');

        return $p && $s ? 'Prioritas Strategis' : ($p ? 'Prioritas' : ($s ? 'Strategis' : 'Belum diklasifikasi'));
    }

    public function getKlasifikasiLabelAttribute(): string
    {
        return self::normalisasiKlasifikasi($this->klasifikasi_pemda);
    }

    /** Daftar sektor yang terdeteksi dari teks potensi_pemda (lihat config/sektor.php). */
    public function getSektorTerdeteksiAttribute(): array
    {
        return \App\Support\Sektor::deteksi((string) $this->potensi_pemda);
    }

    public function getJumlahSektorAttribute(): int
    {
        return count($this->sektor_terdeteksi);
    }

    /** Jumlah tujuan yang tertulis pada kebutuhan_konektivitas (dipisah koma, titik koma, atau baris baru). null bila kosong. */
    public function getJumlahTujuanAttribute(): ?int
    {
        $teks = trim((string) $this->kebutuhan_konektivitas);
        if ($teks === '') {
            return null;
        }
        $butir = array_filter(array_map('trim', preg_split('/[,;\n]+/', $teks)), fn ($x) => $x !== '');

        return count($butir);
    }

    // ---- Tampilan daftar Data Wilayah ----

    /** Pengajuan terbaru (relasi ruteSubsidi sebaiknya sudah di-eager-load). */
    public function getPengajuanTerbaruAttribute(): ?RuteSubsidi
    {
        return $this->ruteSubsidi->sortByDesc('id_rute_subsidi')->first();
    }

    /** Permintaan revisi INTRAM yang belum dipenuhi Pemda. */
    public function getPermintaanTerbukaAttribute()
    {
        return $this->permintaanRevisi->where('status', PermintaanRevisi::MENUNGGU)->values();
    }

    public function getStatusLabelAttribute(): string
    {
        // Ada permintaan revisi dari INTRAM yang belum dipenuhi -> tampil "Perlu Revisi"
        if ($this->permintaan_terbuka->isNotEmpty()) {
            return StatusPengajuan::REVISI;
        }

        return StatusPengajuan::normalisasi($this->pengajuan_terbaru?->status_pengajuan);
    }

    public function getStatusKelasAttribute(): string
    {
        return StatusPengajuan::kelas($this->status_label);
    }

    /** Butir potensi tanpa penomoran: ["Konstruksi", "Pertanian Sagu", ...] */
    public function getPotensiDaftarAttribute(): array
    {
        $butir = array_map(
            fn ($x) => trim(preg_replace('/^\s*\d+\.\s*/', '', trim($x))),
            explode(';', (string) $this->potensi_pemda)
        );

        return array_values(array_filter($butir, fn ($x) => $x !== ''));
    }

    /** Daftar tujuan pada kebutuhan_konektivitas: ["Merauke", "Jayapura", ...] */
    public function getTujuanDaftarAttribute(): array
    {
        $butir = array_map('trim', preg_split('/[,;\n]+/', (string) $this->kebutuhan_konektivitas));

        return array_values(array_filter($butir, fn ($x) => $x !== ''));
    }

    /** Butir keterangan daerah: ["3TP", "KSPEAN", ...] */
    public function getKeteranganDaftarAttribute(): array
    {
        $butir = array_map('trim', explode(',', (string) $this->keterangan_pemda));

        return array_values(array_filter($butir, fn ($x) => $x !== ''));
    }

    public function getPotensiRingkasAttribute(): string
    {
        return implode(', ', $this->potensi_daftar);
    }

    /**
     * Hanya pemilik data yang boleh membatalkan pengajuan, dan tidak setelah disetujui Moda
     * (Verifikasi 1 / Verifikasi Final). Pengajuan yang ditolak atau perlu revisi boleh dibatalkan
     * agar Pemda dapat mengajukan ulang.
     */
    public function bisaDiubahOleh(User $user): bool
    {
        // Pengajuan pada moda mana pun yang sudah disetujui mengunci seluruh data daerah ini
        $adaYangDisetujui = $this->ruteSubsidi->contains(
            fn ($r) => StatusPengajuan::grup($r->status_pengajuan) === 'disetujui'
        );

        return $this->id_user !== null
            && (int) $this->id_user === (int) $user->id_user
            && ! $adaYangDisetujui;
    }

    /**
     * Dokumen pengajuan hanya untuk pemilik data, Super Admin, Intram, dan Moda yang menjadi
     * tujuan salah satu pengajuan daerah ini.
     */
    public function dokumenBisaDiunduhOleh(User $user): bool
    {
        if ($this->id_user !== null && (int) $this->id_user === (int) $user->id_user) {
            return true;
        }
        if ($user->isSuperAdmin() || $user->isIntram()) {
            return true;
        }
        if ($user->isModa() && $user->id_moda !== null) {
            return $this->ruteSubsidi->contains(fn ($r) => (int) $r->id_moda === (int) $user->id_moda);
        }

        return false;
    }

    /**
     * Data Wilayah per role: Pemda hanya data yang diunggahnya; Moda hanya data pemda yang memiliki pengajuan
     * untuk moda akun itu; INTRAM dan Super Admin melihat semuanya.
     */
    public function scopeTerlihatOleh($query, User $user)
    {
        if ($user->isSuperAdmin() || $user->isIntram()) {
            return $query;
        }
        if ($user->isModa()) {
            return $user->id_moda === null
                ? $query->whereRaw('1 = 0')
                : $query->whereHas('ruteSubsidi', fn ($r) => $r->where('id_moda', $user->id_moda));
        }

        return $query->where('data_pemda.id_user', $user->id_user);
    }

    /** Pengajuan terbaru di atas; daerah tanpa pengajuan menyusul berdasarkan id terbesar. */
    public function scopeTerbaru($query)
    {
        return $query
            ->orderByRaw('(SELECT MAX(rute_subsidi.created_at) FROM rute_subsidi WHERE rute_subsidi.id_pemda = data_pemda.id_pemda) DESC')
            ->orderByDesc('id_pemda');
    }

    /**
     * Pemilik data boleh mengajukan berkali-kali. Data tanpa pemilik hanya bisa diambil alih
     * bila belum punya pengajuan; data milik akun lain tidak boleh diajukan.
     */
    public function bisaDiajukanOleh(User $user): bool
    {
        if ($this->id_user !== null) {
            return (int) $this->id_user === (int) $user->id_user;
        }

        return ! $this->ruteSubsidi()->exists();
    }
}

      hasil.forEach(function (k, i) {
        var li = document.createElement('li');
        li.id = 'ac-opt-' + i; li.setAttribute('role', 'option'); li.setAttribute('aria-selected', 'false');
        var a = document.createElement('span'); a.textContent = k.nama;
        var b = document.createElement('small'); b.textContent = k.provinsi;
        li.appendChild(a); li.appendChild(b);
        li.addEventListener('mousedown', function (e) { e.preventDefault(); pilih(k); });   // cegah textarea kehilangan fokus
        list.appendChild(li);
      });
      list.hidden = false;
      ta.setAttribute('aria-expanded', 'true');
    }

    function sorot() {
      $$('li', list).forEach(function (li, i) {
        li.setAttribute('aria-selected', String(i === aktif));
        if (i === aktif) { ta.setAttribute('aria-activedescendant', li.id); li.scrollIntoView({ block: 'nearest' }); }
      });
    }

    function perbarui() {
      var t = tokenAktif();
      if (t.teks.length < 2) { tutup(); return; }
      hasil = cari(t);
      if (!hasil.length) { tutup(); return; }
      aktif = -1;
      gambar();
    }

    function pilih(k) {
      var v = ta.value, t = tokenAktif();
      var depan = v.slice(0, t.mulai).replace(/[ \t]+$/, ''), belakang = v.slice(t.akhir);
      var pemisah = (depan === '' || /\n$/.test(depan)) ? '' : ' ';
      var tambah = belakang.trim() === '' ? ', ' : '';                                // siap mengetik daerah berikutnya
      var awal = depan + pemisah + k.nama + tambah;
      ta.value = awal + belakang;
      ta.setSelectionRange(awal.length, awal.length);
      ta.focus();
      tutup();
    }

    ta.addEventListener('input', perbarui);
    ta.addEventListener('click', perbarui);
    ta.addEventListener('blur', function () { setTimeout(tutup, 120); });
    ta.addEventListener('keydown', function (e) {
      if (list.hidden) return;
      if (e.key === 'ArrowDown') { e.preventDefault(); aktif = (aktif + 1) % hasil.length; sorot(); }
      else if (e.key === 'ArrowUp') { e.preventDefault(); aktif = (aktif - 1 + hasil.length) % hasil.length; sorot(); }
      else if (e.key === 'Enter' && aktif >= 0) { e.preventDefault(); pilih(hasil[aktif]); }
      else if (e.key === 'Escape') { e.stopPropagation(); tutup(); }
    });
  }

  /* ---------- Form user: saran Instansi (kab/kota) dari server ---------- */
  var inInst = $('#instansi'), listInst = $('#instansiList'), idKab = $('#id_kab_kota');
  if (inInst && listInst && idKab && inInst.dataset.url) {
    var instHasil = [], instAktif = -1, instTimer = null, instSeq = 0;

    var tutupInst = function () {
      listInst.hidden = true; listInst.innerHTML = ''; instAktif = -1;
      inInst.setAttribute('aria-expanded', 'false'); inInst.removeAttribute('aria-activedescendant');
    };
    var tandaiInst = function (i) {
      var li = $$('li', listInst);
      li.forEach(function (el, n) { el.setAttribute('aria-selected', n === i ? 'true' : 'false'); });
      instAktif = i;
      if (li[i]) { inInst.setAttribute('aria-activedescendant', li[i].id); li[i].scrollIntoView({ block: 'nearest' }); }
    };
    var pilihInst = function (s) {
      inInst.value = s.value;
      idKab.value = s.id_kab_kota;                         // ID inilah yang dikirim & divalidasi server
      var w = $('#id_wilayah'); if (w && s.id_wilayah) w.value = s.id_wilayah;
      inInst.classList.remove('is-invalid');
      tutupInst();
    };
    var gambarInst = function () {
      listInst.innerHTML = '';
      if (!instHasil.length) { tutupInst(); return; }
      instHasil.forEach(function (s, i) {
        var li = document.createElement('li');
        li.id = 'instansi-opt-' + i; li.setAttribute('role', 'option'); li.setAttribute('aria-selected', 'false');
        var a = document.createElement('span'); a.textContent = s.value;
        var b = document.createElement('small'); b.textContent = s.provinsi;
        li.appendChild(a); li.appendChild(b);
        li.addEventListener('mousedown', function (e) { e.preventDefault(); pilihInst(s); });
        listInst.appendChild(li);
      });
      listInst.hidden = false;
      inInst.setAttribute('aria-expanded', 'true');
    };

    inInst.addEventListener('input', function () {
      idKab.value = '';                                    // teks diubah: pilihan sebelumnya tidak berlaku lagi
      clearTimeout(instTimer);
      var q = inInst.value.trim();
      if (q.length < 2) { instSeq++; tutupInst(); return; }
      instTimer = setTimeout(function () {
        var n = ++instSeq;
        fetch(inInst.dataset.url + '?q=' + encodeURIComponent(q), {
          credentials: 'same-origin',
          headers: { 'Accept': 'application/json', 'X-Requested-With': 'XMLHttpRequest' }
        }).then(function (r) { return r.ok ? r.json() : []; }).then(function (d) {
          if (n !== instSeq) return;                       // abaikan respons yang sudah usang
          instHasil = Array.isArray(d) ? d : []; instAktif = -1; gambarInst();
        }).catch(tutupInst);
      }, 200);
    });
    inInst.addEventListener('keydown', function (e) {
      if (listInst.hidden || !instHasil.length) return;
      if (e.key === 'ArrowDown') { e.preventDefault(); tandaiInst((instAktif + 1) % instHasil.length); }
      else if (e.key === 'ArrowUp') { e.preventDefault(); tandaiInst((instAktif - 1 + instHasil.length) % instHasil.length); }
      else if (e.key === 'Enter' && instAktif >= 0) { e.preventDefault(); pilihInst(instHasil[instAktif]); }
      else if (e.key === 'Escape') { tutupInst(); }
    });
    inInst.addEventListener('blur', function () { setTimeout(tutupInst, 150); });
  }

  /* ---------- Form user: cakupan & hak akses menyesuaikan role ---------- */
  var peran = $('#id_role'), dataRole = $('#data-role');
  if (peran && dataRole) {
    var info = {};
    try { info = JSON.parse(dataRole.textContent); } catch (e) { info = {}; }
    var panel = $('#roleInfo'), daftar = $('#roleInfoList'), catatan = $('#roleInfoCatatan'), tandaWajib = $('#wajibInstansi');

    var sesuaikan = function () {
      var r = info[peran.value];
      var butuh = r ? r.syarat : [];

      $$('[data-butuh]').forEach(function (wadah) {
        var perlu = butuh.indexOf(wadah.dataset.butuh) !== -1;
        wadah.hidden = !perlu;
        $$('input,select', wadah).forEach(function (el) { el.disabled = !perlu; el.required = perlu; });
      });
      $$('[data-wajib]').forEach(function (el) { el.required = butuh.indexOf(el.dataset.wajib) !== -1; });
      if (tandaWajib) tandaWajib.hidden = butuh.indexOf('instansi') === -1;

      if (!r) { panel.hidden = true; return; }
      daftar.innerHTML = '';
      if (r.super) {
        catatan.textContent = 'Super Admin memiliki seluruh hak akses dan dapat mengelola akun.';
      } else {
        r.hak.forEach(function (nama) {
          var li = document.createElement('li');
          li.textContent = nama;
          daftar.appendChild(li);
        });
        catatan.textContent = r.hak.length
          ? 'Hak akses diatur per role lewat tombol "Atur hak akses" di halaman Data User.'
          : 'Role ini belum memiliki hak akses.';
      }
      panel.hidden = false;
    };
    peran.addEventListener('change', sesuaikan);
    sesuaikan();
  }

  /* ---------- Review laporan (Moda): checklist & keputusan ---------- */
  var fk = $('#formKeputusan');
  if (fk) {
    var cek = $$('input[name="periksa[]"]', fk);
    var hitung = $('#cekHitung'), setuju = $('#btnSetuju'), catatan = $('#catatan');

    var perbaruiCek = function () {
      var n = cek.filter(function (c) { return c.checked; }).length;
      if (hitung) {
        hitung.textContent = n + ' dari ' + cek.length + ' pemeriksaan telah dikonfirmasi';
        hitung.classList.toggle('ok', cek.length > 0 && n === cek.length);
      }
      if (setuju) setuju.disabled = n !== cek.length;       // Setujui hanya aktif bila semua pemeriksaan dikonfirmasi
    };
    cek.forEach(function (c) { c.addEventListener('change', perbaruiCek); });
    if (cek.length) perbaruiCek();

    // Catatan wajib untuk Minta Revisi / Tolak
    ['btnRevisi', 'btnTolak'].forEach(function (id) {
      var b = document.getElementById(id);
      if (b && catatan) b.addEventListener('click', function () { catatan.required = true; });
    });
    $$('[data-wajib-catatan]', fk).forEach(function (b) {
      if (catatan) b.addEventListener('click', function () { catatan.required = true; });
    });
    if (setuju && catatan) setuju.addEventListener('click', function () { catatan.required = false; });

    fk.addEventListener('submit', function (e) {
      var v = e.submitter ? e.submitter.value : '';
      var opsi = {
        setuju:    { judul: 'Setujui laporan ini?', pesan: 'Laporan akan ditandai disetujui dan diteruskan ke tahap berikutnya.', ok: 'Ya, setujui', nada: 'primary' },
        revisi:    { judul: 'Minta revisi laporan?', pesan: 'Laporan dikembalikan kepada pengirim untuk diperbaiki sesuai catatan Anda.', ok: 'Ya, minta revisi', nada: 'warning' },
        perbaikan: { judul: 'Kembalikan surat?', pesan: 'Surat dikembalikan untuk diperbaiki sesuai catatan Anda.', ok: 'Ya, kembalikan', nada: 'warning' },
        tolak:     { judul: 'Tolak laporan ini?', pesan: 'Laporan akan ditolak dan pengirim diberi tahu.', ok: 'Ya, tolak', nada: 'danger' }
      }[v];
      if (opsi) {
        opsi.catatan = 'Keputusan akan tercatat permanen pada riwayat persetujuan.';
        konfirmasiSubmit(e, fk, opsi);
      }
    });
  }

  /* ---------- Surat CHR: unggah, ringkasan, dan kirim ---------- */
  var fs = $('#formSurat');
  if (fs) {
    var MAKS_BYTE = (parseInt(fs.dataset.maksMb, 10) || 10) * 1024 * 1024;
    var slots = $$('.sr-slot', fs);
    var baru = {};   // kode slot -> File yang baru dipilih (belum disimpan)

    var fmt = function (b) {
      return b >= 1048576 ? (b / 1048576).toFixed(1).replace('.', ',') + ' MB' : Math.max(1, Math.round(b / 1024)) + ' KB';
    };
    var adaDiServer = function (slot) { return slot.dataset.adaNama !== undefined; };

    var perbaruiSlot = function (slot) {
      var f = baru[slot.dataset.slot], ada = adaDiServer(slot);
      var fAda = $('.sr-file-ada', slot), fBaru = $('.sr-file-baru', slot), drop = $('.sr-drop', slot), st = $('[data-state]', slot);
      fAda.hidden = !(ada && !f);
      fBaru.hidden = !f;
      drop.hidden = !!(ada || f);
      if (f) {
        $('[data-nama]', fBaru).textContent = f.name;
        $('[data-meta]', fBaru).textContent = 'PDF Â· ' + fmt(f.size) + ' Â· Belum disimpan';
      }
      st.textContent = f ? 'Siap disimpan' : (ada ? 'Terunggah' : 'Belum diunggah');
      st.classList.toggle('is-ok', !!(f || ada));
    };

    var hitung = function () {
      var n = 0, total = 0, wajib = 0, wajibOk = 0, tanpaNomor = 0;
      slots.forEach(function (s) {
        var f = baru[s.dataset.slot], ada = adaDiServer(s), wj = s.dataset.wajib === '1';
        var ukuran = f ? f.size : (ada ? (parseInt(s.dataset.adaUkuran, 10) || 0) : 0);
        var ada_ = !!(f || ada);
        if (ada_) { n++; total += ukuran; }
        var nomorEl = $('.sr-nomor input', s), nomorOk = !nomorEl || nomorEl.value.trim() !== '';
        if (ada_ && !nomorOk) tanpaNomor++;
        if (wj) { wajib++; if (ada_ && nomorOk) wajibOk++; }
        var li = $('#srList li[data-slot="' + s.dataset.slot + '"]');
        if (li) {
          li.classList.toggle('ok', ada_ && nomorOk);
          var ket = $('[data-ket]', li);
          if (ket) ket.textContent = ada_ ? (nomorOk ? '' : 'Nomor surat belum diisi') : 'Belum diunggah Â· ' + (wj ? 'Wajib' : 'Opsional');
        }
      });

      var lengkap = wajibOk === wajib && tanpaNomor === 0;
      $('#srCount').textContent = n + ' surat Â· Total ' + (total ? fmt(total) : '0 KB');
      $('#srBar').style.width = (wajib ? Math.round(wajibOk / wajib * 100) : 100) + '%';
      $('#srBarLabel').textContent = wajibOk === wajib ? 'Surat wajib lengkap' : 'Surat wajib belum lengkap (' + wajibOk + '/' + wajib + ')';
      $('#srSiap').classList.toggle('is-belum', !lengkap);
      $('#srSiapChip').textContent = lengkap ? 'Siap dikirim' : 'Belum lengkap';
      $('#srSiapTeks').textContent = lengkap ? 'Surat sudah siap dikirim' : (wajibOk < wajib ? 'Lengkapi ' + (wajib - wajibOk) + ' surat wajib (berkas dan nomor surat)' : 'Isi nomor surat pada setiap berkas');
      $('#srFooterJudul').textContent = lengkap ? 'Siap mengirim ' + n + ' surat ke User INTRAM' : 'Lengkapi surat wajib untuk mengirim';
      $('#btnKirim').disabled = !lengkap;
    };

    slots.forEach(function (slot) {
      var kode = slot.dataset.slot, input = $('input[type=file]', slot), drop = $('.sr-drop', slot);

      var terima = function (file) {
        if (!file) return;
        if (!/\.pdf$/i.test(file.name) || (file.type && file.type !== 'application/pdf')) {
          window.alert('Berkas harus berformat PDF.'); input.value = ''; return;
        }
        if (file.size > MAKS_BYTE) {
          window.alert('Ukuran berkas maksimal ' + (MAKS_BYTE / 1048576) + ' MB.'); input.value = ''; return;
        }
        baru[kode] = file;
        perbaruiSlot(slot); hitung();
      };

      $$('[data-pilih],[data-ganti]', slot).forEach(function (b) { b.addEventListener('click', function () { input.click(); }); });
      input.addEventListener('change', function () {
        if (input.files && input.files[0]) { terima(input.files[0]); }
        else { delete baru[kode]; perbaruiSlot(slot); hitung(); }
      });
      $$('[data-batal]', slot).forEach(function (b) {
        b.addEventListener('click', function () { input.value = ''; delete baru[kode]; perbaruiSlot(slot); hitung(); });
      });

      // Tarik dan lepas berkas
      ['dragenter', 'dragover'].forEach(function (ev) {
        drop.addEventListener(ev, function (e) { e.preventDefault(); drop.classList.add('is-over'); });
      });
      drop.addEventListener('dragleave', function () { drop.classList.remove('is-over'); });
      drop.addEventListener('drop', function (e) {
        e.preventDefault(); drop.classList.remove('is-over');
        var f = e.dataTransfer && e.dataTransfer.files[0];
        if (!f) return;
        try { var dt = new DataTransfer(); dt.items.add(f); input.files = dt.files; }
        catch (err) { window.alert('Browser Anda belum mendukung tarik-dan-lepas. Gunakan tombol Pilih berkas.'); return; }
        terima(f);
      });
    });

    var ta = $('#srCatatan'), ctr = $('#srCounter');
    if (ta && ctr) {
      var hitungKarakter = function () { ctr.textContent = ta.value.length.toLocaleString('id-ID') + ' / 1.000 karakter'; };
      ta.addEventListener('input', hitungKarakter);
      hitungKarakter();
    }

    fs.addEventListener('submit', function (e) {
      var v = e.submitter ? e.submitter.value : '';
      if (v === 'kirim') konfirmasiSubmit(e, fs, { judul: 'Kirim surat ke User INTRAM?', pesan: 'Surat yang sudah dikirim tidak dapat diubah lagi.', ok: 'Ya, kirim', nada: 'primary' });
    });

    $$('.sr-nomor input', fs).forEach(function (el) { el.addEventListener('input', hitung); });
    slots.forEach(perbaruiSlot);
    hitung();
  }

  /* ---------- Upload surat ---------- */
  var MAX = 5 * 1024 * 1024;
  $$('.btn-upload, .upload-name').forEach(function (el) {
    el.addEventListener('click', function () {
      var f = document.getElementById(el.dataset.target || el.dataset.for);
      if (f) f.click();
    });
  });
  $$('input[type=file][data-upload]').forEach(function (f) {
    f.addEventListener('change', function () {
      var label = $('.upload-name[data-for="' + f.id + '"]');
      var file = f.files[0];
      var maksMb = parseInt(f.dataset.maksMb, 10) || 5;
      if (file && f.hasAttribute('data-pdf') && !/\.pdf$/i.test(file.name)) {
        alert('Berkas harus berformat PDF.');
        f.value = '';
        label.value = '';
        return;
      }
      if (file && file.size > maksMb * 1024 * 1024) {
        alert('Ukuran file maksimal ' + maksMb + ' MB.');
        f.value = '';
        label.value = '';
        return;
      }
      label.value = file ? file.name : '';
    });
  });

  /* ---------- Ubah kata sandi: tampilkan sandi & cek konfirmasi ---------- */
  (function () {
    var tampil = $('#tampilSandi'), baru = $('#password'), konfirm = $('#password_confirmation'), pesan = $('#sandiBeda');
    if (!tampil || !baru || !konfirm) return;
    var kolom = [$('#password_lama'), baru, konfirm];
    tampil.addEventListener('change', function () {
      kolom.forEach(function (k) { if (k) k.type = tampil.checked ? 'text' : 'password'; });
    });
    var cek = function () {
      var beda = konfirm.value !== '' && baru.value !== konfirm.value;
      konfirm.setCustomValidity(beda ? 'Konfirmasi kata sandi baru tidak sama.' : '');
      if (pesan) pesan.hidden = !beda;
    };
    baru.addEventListener('input', cek);
    konfirm.addEventListener('input', cek);
  })();

  /* ---------- Notifikasi (lonceng) ---------- */
  (function () {
    var btn = $('#notifBtn'), panel = $('#notifPanel');
    if (!btn || !panel) return;
    function tutup() { panel.hidden = true; btn.setAttribute('aria-expanded', 'false'); }
    btn.addEventListener('click', function (e) {
      e.stopPropagation();
      var buka = panel.hidden;
      panel.hidden = !buka;
      btn.setAttribute('aria-expanded', buka ? 'true' : 'false');
    });
    document.addEventListener('click', function (e) { if (!panel.contains(e.target)) tutup(); });
    document.addEventListener('keydown', function (e) { if (e.key === 'Escape') tutup(); });
  })();

  /* ---------- Konfirmasi lewat tombol submit (data-konfirmasi) ---------- */
  $$('form').forEach(function (f) {
    f.addEventListener('submit', function (e) {
      var pesan = e.submitter && e.submitter.dataset ? e.submitter.dataset.konfirmasi : '';
      if (pesan) konfirmasiSubmit(e, f, { judul: 'Konfirmasi', pesan: pesan, ok: 'Ya, lanjutkan', nada: 'warning' });
    });
  });

  /* ---------- Filter otomatis & konfirmasi hapus ---------- */
  $$('[data-autosubmit]').forEach(function (el) {
    el.addEventListener('change', function () { el.form.submit(); });
  });
  $$('form[data-confirm]').forEach(function (f) {
    f.addEventListener('submit', function (e) {
      konfirmasiSubmit(e, f, { judul: 'Hapus data?', pesan: f.dataset.confirm, ok: 'Ya, hapus', nada: 'danger' });
    });
  });

  /* ---------- Cegah submit ganda ---------- */
  var form = $('#formInput');
  if (form) {
    var btn = $('button[type=submit]', form);
    form.addEventListener('submit', function () {
      if (btn) { btn.disabled = true; btn.textContent = 'Menyimpan...'; }
    });
    window.addEventListener('pageshow', function () {
      if (btn) { btn.disabled = false; btn.textContent = 'Tambah Data'; }
      batasiModa();
    });
  }
})();
